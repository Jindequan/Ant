func TestCodexAuditArithmetic(t *testing.T) {
 for _,c:=range []struct{name string;a,b domain.Units;subtract bool;want domain.Units;overflow bool}{
  {"negative-overflow-to-zero",math.MinInt64,math.MinInt64,false,0,true},
  {"legal-subtract-minimum",-1,math.MinInt64,true,math.MaxInt64,false},
  {"minimum-minus-itself",math.MinInt64,math.MinInt64,true,0,false},
 } {t.Run(c.name,func(t *testing.T){var got domain.Units;var err error;if c.subtract{got,err=domain.SubUnits(c.a,c.b)}else{got,err=domain.AddUnits(c.a,c.b)};t.Logf("got=%d err=%v expected=%d overflow=%t",got,err,c.want,c.overflow);if (err!=nil)!=c.overflow||(!c.overflow&&got!=c.want){t.Fail()}})}
}

func TestCodexAuditStaleObservationInput(t *testing.T){
 snap:=domain.ResourceSnapshot{ResourceID:"res-1",WindowKey:"w-1",Version:3,Capacity:100,Committed:30,Adjustment:-10,Reserved:5}
 obs:=&domain.ObservationEvidence{ResourceID:"res-1",WindowKey:"w-1",Remaining:5,CommittedAtStart:30,AdjustmentVersionAtStart:1,ObservedAt:100,ExpiresAt:2000}
 available,err:=resources.CounterAvailable(snap,obs,1000,2);if err!=nil{t.Fatal(err)}
 plan,err:=resources.CounterReserve(domain.ResourceContext{Now:1000,Snapshot:snap,Observation:obs},1)
 t.Logf("authoritativeAdjustmentVersion=2 staleObservationVersion=1 directAvailable=%d reserveAdmitted=%t err=%v",available,plan.Admitted,err)
 if available>0&&!plan.Admitted{t.Error("CounterReserve has no authoritative adjustment version input; substitutes evidence's own version")}
}
func TestCodexAuditCapabilityFingerprint(t *testing.T){
 a:=domain.DomainRequest{Domain:"ai/text",CapabilityProfile:"profile.chat.basic",Payload:json.RawMessage(`{"prompt":"same"}`)}
 b:=a;b.CapabilityProfile="profile.chat.reasoning"
 fingerprint:=func(r domain.DomainRequest)string{f,err:=domain.SemanticFingerprint(domain.FingerprintInput{Domain:r.Domain,PoolID:"pool-1",Payload:r.Payload});if err!=nil{t.Fatal(err)};return f}
 fa,fb:=fingerprint(a),fingerprint(b)
 t.Logf("request A profile=%s B profile=%s fingerprint A=%s B=%s; FingerprintInput lacks profile field",a.CapabilityProfile,b.CapabilityProfile,fa,fb)
 if fa==fb{t.Error("public fingerprint input cannot distinguish capability profile without an extra undocumented payload convention")}
}
